```ruby
# frozen_string_literal: true

module AnomalyDetection
  class HitlQueueQuery
    TABLE = ENV.fetch("ANOMALY_DETECTION_CLICKHOUSE_TABLE", "pbp_productionDB_optimized.anomaly_scores FINAL")
    DEFAULT_TOP_K = ENV.fetch("HITL_TOP_K", 100).to_i
    DEFAULT_BELOW_P50_PCT = ENV.fetch("HITL_BELOW_P50_PCT", 0.20).to_f

    def initialize(params = {})
      @below_p50_pct = clamp_pct(params.fetch(:below_p50_pct, DEFAULT_BELOW_P50_PCT))
      capacity = positive_integer(params[:capacity])
      top_k = positive_integer(params[:top_k]) || DEFAULT_TOP_K
      if capacity
        @top_k_count = (capacity * (1 - @below_p50_pct)).floor
        @below_k_count = capacity - @top_k_count
      else
        @top_k_count = top_k
        @below_k_count = [1, (top_k * @below_p50_pct).ceil].max
      end
    end

    def call
      p50 = fetch_p50
      top_rows = fetch_rows(top_k_sql).map { |r| build_row(r, "top_k") }
      below_rows = fetch_rows(below_p50_sql(p50)).map { |r| build_row(r, "below_p50") }
      { queue: top_rows + below_rows,
        meta: { top_k_count: top_rows.size, below_p50_count: below_rows.size, p50: p50 } }
    end

    private

    def fetch_p50
      result = Analytics::ClickhouseClient.select(p50_sql)
      return 0.5 if result.failure? || result.to_a.empty?
      result.first.fetch("p50", 0.5).to_f
    end

    def fetch_rows(sql)
      result = Analytics::ClickhouseClient.select(sql)
      return [] if result.failure?
      result.to_a
    end

    def p50_sql
      <<~SQL
        SELECT quantile(0.5)(percentile) AS p50
        FROM #{TABLE}
        WHERE scoring_mode = 'shadow_new'
      SQL
    end

    def top_k_sql
      <<~SQL
        SELECT payment_id, facility_id, facility_name, user_id, scored_at,
               payment_created_at, amount_usd, raw_score, percentile, risk_level,
               is_anomaly, model_version, top_factors
        FROM #{TABLE}
        WHERE scoring_mode = 'shadow_new'
        ORDER BY percentile DESC
        LIMIT #{@top_k_count}
      SQL
    end

    def below_p50_sql(p50)
      <<~SQL
        SELECT payment_id, facility_id, facility_name, user_id, scored_at,
               payment_created_at, amount_usd, raw_score, percentile, risk_level,
               is_anomaly, model_version, top_factors
        FROM #{TABLE}
        WHERE scoring_mode = 'shadow_new'
          AND percentile < #{p50}
        ORDER BY rand()
        LIMIT #{@below_k_count}
      SQL
    end

    def build_row(row, source)
      {
        payment_id: row["payment_id"], facility_id: row["facility_id"],
        facility_name: row["facility_name"], user_id: row["user_id"],
        scored_at: row["scored_at"], payment_created_at: row["payment_created_at"],
        amount_usd: row["amount_usd"], raw_score: row["raw_score"],
        percentile: row["percentile"], risk_level: row["risk_level"],
        is_anomaly: row["is_anomaly"], model_version: row["model_version"],
        top_factors: parse_json_array(row["top_factors"]),
        hitl_queue_source: source
      }
    end

    def parse_json_array(json_string)
      return [] if json_string.blank?
      JSON.parse(json_string)
    rescue JSON::ParserError
      []
    end

    def positive_integer(value)
      parsed = Integer(value, exception: false)
      parsed if parsed&.positive?
    end

    def clamp_pct(value)
      v = value.to_f
      v.clamp(0.0, 1.0)
    end
  end
end
```

Estructura sugerida:
```python
"""Fase 0: construcción del baseline congelado v0 (set dorado + métricas post-fix)."""
import json
from datetime import datetime, timezone
import numpy as np
import pandas as pd

VAL_PARQUET = "data/processed/val_features_enriched.parquet"
GOLDEN_OUT = "output/golden_set_v0.parquet"
N_ROWS = 500
MIN_FACILITIES = 20
SEED = 42

def build_golden_set() -> pd.DataFrame:
    df = pd.read_parquet(VAL_PARQUET)
    per_fac = max(1, N_ROWS // MIN_FACILITIES)
    sampled = (
        df.groupby("facility_id", group_keys=False)
        .apply(lambda g: g.sample(min(len(g), per_fac), random_state=SEED))
    )
    if len(sampled) < N_ROWS:
        extra = df.drop(sampled.index).sample(N_ROWS - len(sampled), random_state=SEED)
        sampled = pd.concat([sampled, extra])
    sampled = sampled.reset_index(drop=True)
    assert len(sampled) >= N_ROWS
    assert sampled["facility_id"].nunique() >= MIN_FACILITIES
    sampled.to_parquet(GOLDEN_OUT, index=False)
    return sampled
```
